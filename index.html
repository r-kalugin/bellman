Консольна реалізація алгоритму Беллмана з покроковим виводом лямбда-матриці 

#include "stdafx.h"
#include <iostream>
#include <stdio.h>
#include <conio.h>
#include "windows.h"
using namespace std;

int main()
{
    SetConsoleCP(1251);
    SetConsoleOutputCP(1251);
    int n;
    const float SUP = 10000;

    cout << "Для реалізації алгоритму Беллмана знаходження мінімального шляху \n в заданому зваженому орієнтованому графі побудуйте матрицю ваг цього графа" << endl;
    cout << "Вкажіть порядок матриці ваг графа:" << endl;
    cin >> n;

    float w[100][100];
    float lmd[100][100];
    float l[100];

    /* Заповнення та заміна 0 на SUP (нескінченність) */
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            cin >> w[i][j];
            if (w[i][j] == 0) w[i][j] = SUP;
        }
    }

    /* Початкові значення лямбда */
    lmd[0][0] = 0;
    for (int j = 1; j < n; j++) {
        lmd[0][j] = 0;
        lmd[j][0] = SUP;
    }

    /* Обчислення лямбда(m) для кожного кроку m */
    int coll;
    int index[100][100];
    for (int m = 1; m < n; m++) {
        for (int k = 1; k < n; k++) {
            coll = 0;
            for (int i = 0; i < n; i++) {
                l[i] = (lmd[i][m-1]) + (w[i][k]);
            }
            float min = l[0];
            for (int t = 0; t < n; t++) {
                if (l[t] <= min) {
                    min = l[t];
                    index[k][m] = t + 1;
                }
            }
            lmd[k][m] = min;
        }
    }

    return 0;
}
